This recipe demonstrates how to wire [Chakra UI](https://chakra-ui.com/) components with BlueForm.

Each field type is a small component that reads everything from `useField()` and forwards it to the library's input. The form config, `setupForm` and submit logic are the same as in every other recipe. Type the adapter's props with [`AdapterProps`](../../../guide/core/field-authoring/use-field.mdx#typing-an-adapters-props), so a config's `props` takes the library's own props; see [Integrations](./index.mdx) for the live demos and the event-signature notes.

## Text field adapter

```tsx
// Chakra UI v3 (Field namespace). For v2, use FormControl / FormLabel / FormErrorMessage.
import { Field, Input } from "@chakra-ui/react"
import { type AdapterProps, useField } from "react-headless-form"

// Chakra's Input props minus what useField() controls (value, onChange, disabled, ...)
export type TextFieldProps = AdapterProps<React.ComponentProps<typeof Input>>

export function TextField(props: TextFieldProps) {
  const { value, onChange, onBlur, ref, label, description, errorMessage, required, disabled, readOnly } = useField()
  return (
    <Field.Root invalid={Boolean(errorMessage)} required={required} disabled={disabled} readOnly={readOnly}>
      <Field.Label>
        {label} <Field.RequiredIndicator />
      </Field.Label>
      <Input
        {...props}
        ref={ref}
        value={value ?? ""}
        onChange={(e) => onChange?.(e.target.value)}
        onBlur={onBlur}
      />
      {description && <Field.HelperText>{description}</Field.HelperText>}
      <Field.ErrorText>{errorMessage}</Field.ErrorText>
    </Field.Root>
  )
}
```

Register it in `setupForm`:

```tsx
export const [Form, defineConfig] = setupForm({
  fieldMapping: defineMapping({ text: TextField }),
  renderRoot: ({ children, onSubmit }) => <form onSubmit={onSubmit}>{children}</form>,
})
```
