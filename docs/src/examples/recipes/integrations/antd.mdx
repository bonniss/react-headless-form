This recipe demonstrates how to wire [Ant Design](https://ant.design/) components with BlueForm.

Each field type is a small component that reads everything from `useField()` and forwards it to the library's input. The form config, `setupForm` and submit logic are the same as in every other recipe — see [Integrations](./index.mdx) for the shared sample form and the event-signature notes.

## Text field adapter

```tsx
import { Form, Input, type InputProps } from "antd"
import { useField } from "react-headless-form"

export function TextField(props: InputProps) {
  const { value, onChange, onBlur, ref, label, description, errorMessage, required, disabled, readOnly } = useField()
  return (
    // Form.Item is used for layout only; validation state comes from BlueForm
    <Form.Item
      label={label}
      required={required}
      extra={description}
      validateStatus={errorMessage ? "error" : undefined}
      help={errorMessage}
    >
      <Input
        {...props}
        ref={ref}
        disabled={disabled}
        readOnly={readOnly}
        value={value ?? ""}
        onChange={(e) => onChange?.(e.target.value)}
        onBlur={onBlur}
      />
    </Form.Item>
  )
}
```

Register it in `setupForm`:

```tsx
export const [Form, defineConfig] = setupForm({
  fieldMapping: defineMapping({ text: TextField }),
  renderRoot: ({ children, onSubmit }) => <form onSubmit={onSubmit}>{children}</form>,
})
```
