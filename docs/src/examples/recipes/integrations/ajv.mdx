This recipe demonstrates how to use [Ajv](https://ajv.js.org/) with BlueForm.

```sh
pnpm add ajv ajv-errors @hookform/resolvers
```

Pass the resolver through `formOptions`. Field-level `rules` are disabled while a resolver is set, so keep all validation in the schema — see [Schema validation](../../../guide/core/schema-validation.mdx).

```tsx
import { ajvResolver } from "@hookform/resolvers/ajv"
import type { JSONSchemaType } from "ajv"

type FormData = { name: string; email: string }

const schema: JSONSchemaType<FormData> = {
  type: "object",
  properties: {
    name: { type: "string", minLength: 1, errorMessage: { minLength: "Name is required" } },
    email: { type: "string", format: "email", errorMessage: { format: "Invalid email" } },
  },
  required: ["name", "email"],
  additionalProperties: false,
}

<Form<FormData>
  formOptions={{ resolver: ajvResolver(schema) }}
  config={{
    name: { type: "text", label: "Name" },
    email: { type: "text", label: "Email" },
  }}
  onSubmit={(data) => save(data)}
/>
```
