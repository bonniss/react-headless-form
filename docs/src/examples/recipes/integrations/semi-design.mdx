This recipe demonstrates how to wire [Semi Design](https://semi.design/en-US/) components with BlueForm.

Each field type is a small component that reads everything from `useField()` and forwards it to the library's input. The form config, `setupForm` and submit logic are the same as in every other recipe — see [Integrations](./index.mdx) for the shared sample form and the event-signature notes.

## Text field adapter

```tsx
import { Input } from "@douyinfe/semi-ui"
import { useField } from "react-headless-form"

export function TextField(props: React.ComponentProps<typeof Input>) {
  const { value, onChange, onBlur, label, errorMessage, required, disabled, readOnly } = useField()
  return (
    <label>
      <span>
        {label} {required && "*"}
      </span>
      <Input
        {...props}
        validateStatus={errorMessage ? "error" : "default"}
        disabled={disabled}
        readonly={readOnly}
        value={value ?? ""}
        onChange={(val) => onChange?.(val)} // value first, event second
        onBlur={onBlur}
      />
      {errorMessage && <div className="semi-form-field-error-message">{errorMessage}</div>}
    </label>
  )
}
```

Register it in `setupForm`:

```tsx
export const [Form, defineConfig] = setupForm({
  fieldMapping: defineMapping({ text: TextField }),
  renderRoot: ({ children, onSubmit }) => <form onSubmit={onSubmit}>{children}</form>,
})
```
