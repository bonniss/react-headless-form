This recipe demonstrates how to wire [shadcn/ui ](https://ui.shadcn.com/) components with BlueForm.

Each field type is a small component that reads everything from `useField()` and forwards it to the library's input. The form config, `setupForm` and submit logic are the same as in every other recipe — see [Integrations](./index.mdx) for the shared sample form and the event-signature notes.

## Text field adapter

```tsx
import { Input } from "@/components/ui/input"
import { Label } from "@/components/ui/label"
import { useField } from "react-headless-form"

export function TextField(props: React.ComponentProps<typeof Input>) {
  const { id, value, onChange, onBlur, ref, label, description, errorMessage, required, disabled, readOnly } = useField()
  return (
    <div className="grid gap-2">
      <Label htmlFor={id}>
        {label} {required && "*"}
      </Label>
      <Input
        {...props}
        id={id}
        ref={ref}
        disabled={disabled}
        readOnly={readOnly}
        aria-invalid={Boolean(errorMessage)}
        value={value ?? ""}
        onChange={(e) => onChange?.(e.target.value)}
        onBlur={onBlur}
      />
      {description && <p className="text-sm text-muted-foreground">{description}</p>}
      {errorMessage && <p className="text-sm text-destructive">{errorMessage}</p>}
    </div>
  )
}
```

Register it in `setupForm`:

```tsx
export const [Form, defineConfig] = setupForm({
  fieldMapping: defineMapping({ text: TextField }),
  renderRoot: ({ children, onSubmit }) => <form onSubmit={onSubmit}>{children}</form>,
})
```
